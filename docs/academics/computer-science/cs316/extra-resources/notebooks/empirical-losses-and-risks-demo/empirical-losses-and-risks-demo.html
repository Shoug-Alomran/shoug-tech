Model Training and Evaluation Function

def train_and_evaluate(X, y, test_size):
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size)
    model = LinearRegression()
    model.fit(X_train, y_train)
    y_pred = model.predict(X_test)
    return mean_squared_error(y_test, y_pred)

This function: 1. Splits data into training and test sets 2. Creates and trains a linear regression model 3. Makes predictions on the test set 4. Returns the Mean Squared Error (MSE)

# Function to train the model
def train_and_evaluate(X, y, test_size):
    """Train model and return model, loss, and train/test split"""
    # returns four parameters, x and y is partitioned based on test_size porportion
    # train_test_split() is built-in
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size)
    model = LinearRegression() # Family of function chosen for this model
    model.fit(X_train, y_train) # Performs the training, and returns the model and finds the relationship between x and y train
    y_pred = model.predict(X_test) # Test sample returns all the y values for x_test- we pass x_test bc we want to obtain the corresponding y_test
    # -------------
    # x | y_true
    # x_test -> y_test
    # x_train -> x_train
    # each x_test in the test set corresponds to a y_test & each x_train in the train set corresponds to a y_train
    # and the value it outputs represents predicted outcome of y (y_pred) e.g. y_pred = y = 1.9x + 1.5
    # y = y_true + noise
    # usually training set error is less relative to testing set error

    train_loss = mean_squared_error(y_train, model.predict(X_train)) # Finds relationship between x and y and outputs the model estimating the parameters m and b
    test_loss = mean_squared_error(y_test, y_pred) # Evaluates the model performance (feeding it with new input (test set))
    # Predicts loss on the trained and test set
    return model, train_loss, test_loss, (X_train, X_test, y_train, y_test)
def plot_model_fit(X, y, y_true, model, split_data, noise_level, n_samples):
    """Plot the data, ground truth, and model predictions"""
    X_train, X_test, y_train, y_test = split_data

    plt.figure(figsize=(6, 4))

    # Plot training data
    # Compares training and testing set
    plt.scatter(X_train, y_train, color='blue', alpha=0.5, label='Training Data')
    plt.scatter(X_test, y_test, color='green', alpha=0.5, label='Test Data')

    # Plot ground truth line
    X_sorted = np.sort(X, axis=0)
    plt.plot(X_sorted, 2 * X_sorted + 1, 'r--', label='Ground Truth (y = 2x + 1)', linewidth=2)

    # Plot predicted line - evaluates the model's learnt relationship
    y_pred_line = model.predict(X_sorted)
    plt.plot(X_sorted, y_pred_line, 'g-', label=f'Predicted (y = {model.coef_[0]:.2f}x + {model.intercept_:.2f})', linewidth=2)

    plt.xlabel('X')
    plt.ylabel('y')
    plt.title(f'Model Fit (Noise={noise_level}, Samples={n_samples})\nMSE = {mean_squared_error(y_test, model.predict(X_test)):.4f}')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

def calculate_empirical_risk(losses, parameter_name):
    """
    Calculate average loss (empirical risk) for each parameter value

    Args:
        losses: List of dictionaries containing 'loss' and parameter values
        parameter_name: Name of parameter to group by (e.g., 'n_samples', 'noise')

    Returns:
        Dictionary mapping parameter values to their average loss
    """
    # Get all unique values of the parameter
    parameter_values = set(loss[parameter_name] for loss in losses)
    # parameter_values = (20,50)

    # Calculate average loss for each parameter value
    average_losses = {}
    for value in parameter_values: # Iterates through param values e.g. n_sample_size=20
        # Get all losses for this parameter value
        losses_for_value = [] # loss value per parameter value
        for loss_dict in losses:
            if loss_dict[parameter_name] == value:
                losses_for_value.append(loss_dict['loss'])

        # Calculate average loss
        average_loss = sum(losses_for_value) / len(losses_for_value)
        average_losses[value] = average_loss

    return average_losses
